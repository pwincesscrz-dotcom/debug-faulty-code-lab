📝 Step 2.1: Syntax Errors SYNTAX ERROR

Syntax errors occur when the code violates JavaScript language rules.

❌ Faulty Code 1:

// Missing closing parenthesis and bracket
function greet(name {
  console.log("Hello, " + name;
}
greet("John");

✅ Fixed Code:

// Fixed: Added missing parenthesis and closing bracket
function greet(name) {
  console.log("Hello, " + name);
}
greet("John");

❌ Faulty Code 2:

// Missing quotes and comma
let colors = [red, green, blue];
console.log(colors);

✅ Fixed Code:

// Fixed: Added quotes around strings
let colors = ["red", "green", "blue"];
console.log(colors);

⚡ Step 2.2: Runtime Errors RUNTIME ERROR

Runtime errors occur when the code executes but encounters an issue.

❌ Faulty Code 1:

// Accessing undefined variable or property
let user = null;
console.log(user.name);

✅ Fixed Code:

// Fixed: Added null check using optional chaining
let user = null;
console.log(user?.name || "User is null");
// Or check before accessing
if (user) { console.log(user.name); }
else { console.log("User is null"); }

❌ Faulty Code 2:

// Calling non-existent function
calculateTotal(100, 50);

✅ Fixed Code:

// Fixed: Define function before calling
function calculateTotal(price, discount) {
  return price - discount;
}
console.log(calculateTotal(100, 50));

🧠 Step 3: Logical Errors LOGICAL ERROR

Logical errors occur when code runs but produces incorrect results.

❌ Faulty Code 1:

// Incorrect calculation (should be sum, not concatenation)
function calculateSum(a, b) {
  return a + b;
}
console.log(calculateSum(5, "10"));

✅ Fixed Code:

// Fixed: Convert to numbers before adding
function calculateSum(a, b) {
  return Number(a) + Number(b);
}
console.log(calculateSum(5, "10"));

❌ Faulty Code 2:

// Wrong condition logic (should check both conditions)
let age = 25;
let hasLicense = false;
if (age >= 18 || hasLicense) {
  console.log("Can drive");
} else {
  console.log("Cannot drive");
}

✅ Fixed Code:

// Fixed: Use AND (&&) instead of OR (||)
let age = 25;
let hasLicense = false;
if (age >= 18 && hasLicense) {
  console.log("Can drive");
} else {
  console.log("Cannot drive");
}

❌ Faulty Code 3: Infinite Loop

// Missing increment in loop
let i = 0;
while (i < 5) {
  console.log(i);
  // i++ is missing
}

✅ Fixed Code:

// Fixed: Added increment
let i = 0;
while (i < 5) {
  console.log(i);
  i++;
}

🎯 Step 5: Comprehensive Debugging Challenge

❌ Faulty Program:

// This program should calculate average of numbers
// But it has multiple bugs!

function getAverage(numbers) {
  let sum = 0;
  for (let i = 0; i <= numbers.length; i++) {
    sum += numbers[i];
  }
  return sum / numbers.length;
}

let scores = [85, 90, 78, 92, 88];
let average = getAverage(scores);
console.log("Average score:", average);

✅ Fixed Program:

// Fixed version with proper error handling
function getAverage(numbers) {
  // Check if input is valid
  if (!Array.isArray(numbers) || numbers.length === 0) {
    return 0;
  }
  
  let sum = 0;
  // Fixed: Use < instead of <=
  for (let i = 0; i < numbers.length; i++) {
    // Convert to number in case of strings
    sum += Number(numbers[i]);
  }
  return sum / numbers.length;
}

let scores = [85, 90, 78, 92, 88];
let average = getAverage(scores);
console.log("Average score:", average);
